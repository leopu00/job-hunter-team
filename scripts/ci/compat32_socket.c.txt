/*
 * 32-bit code against the broker's seccomp profile (CI only).
 *
 * Built static, without libc, for i386 (int 0x80 on x86_64) and for armhf
 * (AArch32 on arm64). It tries to open AF_ALG (38) and AF_VSOCK (40), each
 * through socket() and through socketcall(SYS_SOCKET), whose family sits
 * behind a pointer that no seccomp rule can read. The exit code says what
 * opened: 16 plus one bit per way (1 socketcall AF_ALG, 2 socket AF_ALG,
 * 4 socketcall AF_VSOCK, 8 socket AF_VSOCK). 16 = all refused. If even exit
 * is refused, it traps: a signal, also a refusal.
 *
 * scripts/ci/compat32_gate.py runs it with and without the profile.
 * Kept as .c.txt, since the repository admits no .c file; the workflow
 * compiles it with `gcc -x c`.
 */

#define AF_ALG 38
#define AF_VSOCK 40
#define SOCK_STREAM 1
#define SOCK_SEQPACKET 5
#define SYS_SOCKET 1

#if defined(__i386__)
#define NR_EXIT 1
#define NR_SOCKETCALL 102
#define NR_SOCKET 359

static long sys3(long nr, long a, long b, long c)
{
    long ret;
    __asm__ volatile("int $0x80" : "=a"(ret) : "a"(nr), "b"(a), "c"(b), "d"(c) : "memory");
    return ret;
}
#elif defined(__arm__)
#define NR_EXIT 1
#define NR_SOCKETCALL 102
#define NR_SOCKET 281

static long sys3(long nr, long a, long b, long c)
{
    register long r7 __asm__("r7") = nr;
    register long r0 __asm__("r0") = a;
    register long r1 __asm__("r1") = b;
    register long r2 __asm__("r2") = c;
    __asm__ volatile("svc 0" : "+r"(r0) : "r"(r7), "r"(r1), "r"(r2) : "memory");
    return r0;
}
#else
#error "32-bit i386 or arm only"
#endif

static int opened(long fd)
{
    return fd >= 0;
}

static int by_socketcall(long family, long type)
{
    long args[3] = {family, type, 0};
    return opened(sys3(NR_SOCKETCALL, SYS_SOCKET, (long)args, 0));
}

static int by_socket(long family, long type)
{
    return opened(sys3(NR_SOCKET, family, type, 0));
}

void _start(void)
{
    long code = 16;
    code += by_socketcall(AF_ALG, SOCK_SEQPACKET) ? 1 : 0;
    code += by_socket(AF_ALG, SOCK_SEQPACKET) ? 2 : 0;
    code += by_socketcall(AF_VSOCK, SOCK_STREAM) ? 4 : 0;
    code += by_socket(AF_VSOCK, SOCK_STREAM) ? 8 : 0;
    sys3(NR_EXIT, code, 0, 0);
    __builtin_trap();
}
